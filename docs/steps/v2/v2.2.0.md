### Generate Registration Options

```typescript
// app/api/webauthn/register/options/route.ts
import { auth } from "@/lib/auth"
import { generateRegistrationOptions } from "@simplewebauthn/server"
import { prisma } from "@/lib/prisma"

const rpName = process.env.WEBAUTHN_RP_NAME || "WalletWise"
const rpID = process.env.WEBAUTHN_RP_ID || "localhost"

export async function POST(request: Request) {
  const session = await auth()
  if (!session?.user?.id) {
    return Response.json({ error: "No autorizado" }, { status: 401 })
  }

  const user = await prisma.user.findUnique({
    where: { id: session.user.id },
    include: { authenticators: true },
  })

  if (!user) {
    return Response.json({ error: "Usuario no encontrado" }, { status: 404 })
  }

  const options = await generateRegistrationOptions({
    rpName,
    rpID,
    userID: user.id,
    userName: user.email,
    userDisplayName: user.name || user.email,
    excludeCredentials: user.authenticators.map(auth => ({
      id: Buffer.from(auth.credentialID, "base64url"),
      type: "public-key",
      transports: auth.transports ? JSON.parse(auth.transports) : undefined,
    })),
    authenticatorSelection: {
      residentKey: "preferred",
      userVerification: "preferred",
    },
  })

  // Guardar challenge en session/cache para verificacion
  // En produccion, usar Redis o similar
  await prisma.user.update({
    where: { id: user.id },
    data: {
      // Guardar temporalmente - en produccion usar cache
      // currentChallenge: options.challenge
    },
  })

  return Response.json(options)
}
```

### Verify Registration

```typescript
// app/api/webauthn/register/verify/route.ts
import { auth } from "@/lib/auth"
import { verifyRegistrationResponse } from "@simplewebauthn/server"
import { prisma } from "@/lib/prisma"

const rpID = process.env.WEBAUTHN_RP_ID || "localhost"
const origin = process.env.WEBAUTHN_ORIGIN || "http://localhost:3000"

export async function POST(request: Request) {
  const session = await auth()
  if (!session?.user?.id) {
    return Response.json({ error: "No autorizado" }, { status: 401 })
  }

  const body = await request.json()
  const { response, name } = body

  const user = await prisma.user.findUnique({
    where: { id: session.user.id },
  })

  if (!user) {
    return Response.json({ error: "Usuario no encontrado" }, { status: 404 })
  }

  try {
    const verification = await verifyRegistrationResponse({
      response,
      expectedChallenge: /* obtener de cache */,
      expectedOrigin: origin,
      expectedRPID: rpID,
    })

    if (verification.verified && verification.registrationInfo) {
      const { credentialPublicKey, credentialID, counter } = verification.registrationInfo

      await prisma.authenticator.create({
        data: {
          userId: user.id,
          credentialID: Buffer.from(credentialID).toString("base64url"),
          credentialPublicKey: Buffer.from(credentialPublicKey).toString("base64url"),
          counter: BigInt(counter),
          credentialDeviceType: verification.registrationInfo.credentialDeviceType,
          credentialBackedUp: verification.registrationInfo.credentialBackedUp,
          transports: response.response.transports
            ? JSON.stringify(response.response.transports)
            : null,
          name: name || "Passkey",
        },
      })

      // Activar WebAuthn si es el primer authenticator
      await prisma.user.update({
        where: { id: user.id },
        data: { webauthnEnabled: true },
      })

      return Response.json({ verified: true })
    }

    return Response.json({ verified: false }, { status: 400 })
  } catch (error) {
    return Response.json({ error: "Error de verificacion" }, { status: 400 })
  }
}
```

## WebAuthn - Autenticacion

### Generate Authentication Options

```typescript
// app/api/webauthn/authenticate/options/route.ts
import { generateAuthenticationOptions } from "@simplewebauthn/server"
import { prisma } from "@/lib/prisma"

const rpID = process.env.WEBAUTHN_RP_ID || "localhost"

export async function POST(request: Request) {
  const { email } = await request.json()

  const user = await prisma.user.findUnique({
    where: { email },
    include: { authenticators: true },
  })

  if (!user || user.authenticators.length === 0) {
    return Response.json({ error: "Usuario sin passkeys" }, { status: 400 })
  }

  const options = await generateAuthenticationOptions({
    rpID,
    allowCredentials: user.authenticators.map(auth => ({
      id: Buffer.from(auth.credentialID, "base64url"),
      type: "public-key",
      transports: auth.transports ? JSON.parse(auth.transports) : undefined,
    })),
    userVerification: "preferred",
  })

  return Response.json(options)
}
```

### Verify Authentication

```typescript
// app/api/webauthn/authenticate/verify/route.ts
import { verifyAuthenticationResponse } from "@simplewebauthn/server"
import { prisma } from "@/lib/prisma"
import { signIn } from "@/lib/auth"

const rpID = process.env.WEBAUTHN_RP_ID || "localhost"
const origin = process.env.WEBAUTHN_ORIGIN || "http://localhost:3000"

export async function POST(request: Request) {
  const { email, response } = await request.json()

  const user = await prisma.user.findUnique({
    where: { email },
    include: { authenticators: true },
  })

  if (!user) {
    return Response.json({ error: "Usuario no encontrado" }, { status: 404 })
  }

  const authenticator = user.authenticators.find(
    auth => auth.credentialID === response.id
  )

  if (!authenticator) {
    return Response.json({ error: "Authenticator no encontrado" }, { status: 400 })
  }

  try {
    const verification = await verifyAuthenticationResponse({
      response,
      expectedChallenge: /* obtener de cache */,
      expectedOrigin: origin,
      expectedRPID: rpID,
      authenticator: {
        credentialID: Buffer.from(authenticator.credentialID, "base64url"),
        credentialPublicKey: Buffer.from(authenticator.credentialPublicKey, "base64url"),
        counter: Number(authenticator.counter),
      },
    })

    if (verification.verified) {
      // Actualizar counter
      await prisma.authenticator.update({
        where: { id: authenticator.id },
        data: { counter: BigInt(verification.authenticationInfo.newCounter) },
      })

      // Crear session
      // Nota: Esto requiere un provider custom en Auth.js
      return Response.json({
        verified: true,
        userId: user.id,
      })
    }

    return Response.json({ verified: false }, { status: 400 })
  } catch (error) {
    return Response.json({ error: "Error de verificacion" }, { status: 400 })
  }
}
```

## 2FA TOTP

### Setup (Generar QR)

```typescript
// app/api/2fa/totp/setup/route.ts
import { auth } from "@/lib/auth"
import { authenticator } from "otplib"
import QRCode from "qrcode"
import { prisma } from "@/lib/prisma"

export async function POST(request: Request) {
  const session = await auth()
  if (!session?.user?.id) {
    return Response.json({ error: "No autorizado" }, { status: 401 })
  }

  const user = await prisma.user.findUnique({
    where: { id: session.user.id },
  })

  if (!user) {
    return Response.json({ error: "Usuario no encontrado" }, { status: 404 })
  }

  if (user.totpEnabled) {
    return Response.json({ error: "2FA ya esta activado" }, { status: 400 })
  }

  // Generar secret
  const secret = authenticator.generateSecret()

  // Guardar secret temporalmente (sin activar)
  await prisma.user.update({
    where: { id: user.id },
    data: { totpSecret: secret },
  })

  // Generar OTP URI
  const otpauth = authenticator.keyuri(
    user.email,
    "WalletWise",
    secret
  )

  // Generar QR code
  const qrCode = await QRCode.toDataURL(otpauth)

  return Response.json({
    secret,
    qrCode,
  })
}
```

### Verify TOTP

```typescript
// app/api/2fa/totp/verify/route.ts
import { auth } from "@/lib/auth"
import { authenticator } from "otplib"
import { prisma } from "@/lib/prisma"

export async function POST(request: Request) {
  const session = await auth()
  if (!session?.user?.id) {
    return Response.json({ error: "No autorizado" }, { status: 401 })
  }

  const { code } = await request.json()

  const user = await prisma.user.findUnique({
    where: { id: session.user.id },
  })

  if (!user || !user.totpSecret) {
    return Response.json({ error: "2FA no configurado" }, { status: 400 })
  }

  const isValid = authenticator.verify({
    token: code,
    secret: user.totpSecret,
  })

  if (!isValid) {
    return Response.json({ error: "Codigo invalido" }, { status: 400 })
  }

  // Activar 2FA si es primera verificacion
  if (!user.totpEnabled) {
    await prisma.user.update({
      where: { id: user.id },
      data: { totpEnabled: true },
    })
  }

  return Response.json({ verified: true })
}
```

### Disable 2FA

```typescript
// app/api/2fa/totp/disable/route.ts
import { auth } from "@/lib/auth"
import { authenticator } from "otplib"
import { prisma } from "@/lib/prisma"

export async function POST(request: Request) {
  const session = await auth()
  if (!session?.user?.id) {
    return Response.json({ error: "No autorizado" }, { status: 401 })
  }

  const { code, password } = await request.json()

  const user = await prisma.user.findUnique({
    where: { id: session.user.id },
  })

  if (!user || !user.totpSecret || !user.totpEnabled) {
    return Response.json({ error: "2FA no esta activado" }, { status: 400 })
  }

  // Verificar codigo TOTP
  const isValid = authenticator.verify({
    token: code,
    secret: user.totpSecret,
  })

  if (!isValid) {
    return Response.json({ error: "Codigo invalido" }, { status: 400 })
  }

  // Desactivar 2FA
  await prisma.user.update({
    where: { id: user.id },
    data: {
      totpEnabled: false,
      totpSecret: null,
    },
  })

  return Response.json({ success: true })
}
```

### WebAuthn Register

```typescript
// components/auth/webauthn-register.tsx
"use client"

import { startRegistration } from "@simplewebauthn/browser"
import { Button } from "@/components/ui/button"
import { Fingerprint } from "lucide-react"

export function WebAuthnRegister() {
  async function handleRegister() {
    try {
      // Obtener opciones
      const optionsRes = await fetch("/api/webauthn/register/options", {
        method: "POST",
      })
      const options = await optionsRes.json()

      // Iniciar registro con el dispositivo
      const credential = await startRegistration(options)

      // Verificar con el servidor
      const verifyRes = await fetch("/api/webauthn/register/verify", {
        method: "POST",
        body: JSON.stringify({
          response: credential,
          name: "Mi Passkey",
        }),
      })

      const result = await verifyRes.json()

      if (result.verified) {
        // Exito
      }
    } catch (error) {
      console.error("Error registrando passkey:", error)
    }
  }

  return (
    <Button onClick={handleRegister} variant="outline">
      <Fingerprint className="mr-2 h-4 w-4" />
      Agregar Passkey
    </Button>
  )
}
```

### TOTP Setup

```typescript
// components/auth/totp-setup.tsx
"use client"

import { useState } from "react"
import Image from "next/image"
import { Button } from "@/components/ui/button"
import { Input } from "@/components/ui/input"

export function TOTPSetup() {
  const [qrCode, setQrCode] = useState<string | null>(null)
  const [secret, setSecret] = useState<string | null>(null)
  const [code, setCode] = useState("")

  async function handleSetup() {
    const res = await fetch("/api/2fa/totp/setup", { method: "POST" })
    const data = await res.json()
    setQrCode(data.qrCode)
    setSecret(data.secret)
  }

  async function handleVerify() {
    const res = await fetch("/api/2fa/totp/verify", {
      method: "POST",
      body: JSON.stringify({ code }),
    })
    const result = await res.json()
    if (result.verified) {
      // 2FA activado
    }
  }

  if (!qrCode) {
    return (
      <Button onClick={handleSetup}>
        Configurar 2FA
      </Button>
    )
  }

  return (
    <div className="space-y-4">
      <p>Escanea este codigo QR con Google Authenticator:</p>
      <Image src={qrCode} alt="QR Code" width={200} height={200} />
      <p className="text-sm text-muted-foreground">
        Codigo manual: {secret}
      </p>
      <div className="flex gap-2">
        <Input
          value={code}
          onChange={(e) => setCode(e.target.value)}
          placeholder="Ingresa el codigo"
          maxLength={6}
        />
        <Button onClick={handleVerify}>Verificar</Button>
      </div>
    </div>
  )
}
```

## Flujo de Login con 2FA

```
1. Usuario ingresa email/password
2. Verificar credenciales
3. Si 2FA activado:
   a. Redirigir a /verify-2fa
   b. Usuario ingresa codigo TOTP
   c. Verificar codigo
   d. Crear session
4. Si 2FA no activado:
   a. Crear session
5. Redirigir a /dashboard
```
