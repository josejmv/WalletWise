```typescript
// app/api/user/export/route.ts
import { auth } from "@/lib/auth"
import { prisma } from "@/lib/prisma"
import { NextResponse } from "next/server"

export async function GET() {
  const session = await auth()
  if (!session?.user?.id) {
    return NextResponse.json({ error: "No autorizado" }, { status: 401 })
  }

  const userId = session.user.id

  // Obtener todos los datos del usuario
  const [
    user,
    accounts,
    jobs,
    incomes,
    expenses,
    transfers,
    budgets,
    budgetContributions,
    categories,
    inventoryCategories,
    inventoryItems,
    inventoryPriceHistory,
    userConfig,
  ] = await Promise.all([
    prisma.user.findUnique({
      where: { id: userId },
      select: { id: true, email: true, name: true, createdAt: true },
    }),
    prisma.account.findMany({ where: { userId } }),
    prisma.job.findMany({ where: { userId } }),
    prisma.income.findMany({ where: { userId } }),
    prisma.expense.findMany({ where: { userId } }),
    prisma.transfer.findMany({ where: { userId } }),
    prisma.budget.findMany({ where: { userId } }),
    prisma.budgetContribution.findMany({ where: { userId } }),
    prisma.category.findMany({ where: { userId } }),
    prisma.inventoryCategory.findMany({ where: { userId } }),
    prisma.inventoryItem.findMany({ where: { userId } }),
    prisma.inventoryPriceHistory.findMany({ where: { userId } }),
    prisma.userConfig.findUnique({ where: { userId } }),
  ])

  const exportData = {
    exportedAt: new Date().toISOString(),
    version: "2.6.0",
    user,
    data: {
      accounts,
      jobs,
      incomes,
      expenses,
      transfers,
      budgets,
      budgetContributions,
      categories,
      inventoryCategories,
      inventoryItems,
      inventoryPriceHistory,
      userConfig,
    },
  }

  return new NextResponse(JSON.stringify(exportData, null, 2), {
    headers: {
      "Content-Type": "application/json",
      "Content-Disposition": `attachment; filename="walletwise-export-${new Date().toISOString().split("T")[0]}.json"`,
    },
  })
}
```

## API: Delete Account

```typescript
// app/api/user/delete/route.ts
import { auth, signOut } from "@/lib/auth"
import { prisma } from "@/lib/prisma"
import bcrypt from "bcryptjs"
import { NextResponse } from "next/server"
import { z } from "zod"

const deleteSchema = z.object({
  password: z.string().min(1, "Password requerido"),
  confirmation: z.literal("ELIMINAR MI CUENTA"),
})

export async function POST(request: Request) {
  const session = await auth()
  if (!session?.user?.id) {
    return NextResponse.json({ error: "No autorizado" }, { status: 401 })
  }

  try {
    const body = await request.json()
    const { password, confirmation } = deleteSchema.parse(body)

    // Verificar password
    const user = await prisma.user.findUnique({
      where: { id: session.user.id },
    })

    if (!user) {
      return NextResponse.json({ error: "Usuario no encontrado" }, { status: 404 })
    }

    // Si tiene password (no es OAuth puro)
    if (user.password) {
      const isValid = await bcrypt.compare(password, user.password)
      if (!isValid) {
        return NextResponse.json({ error: "Password incorrecto" }, { status: 400 })
      }
    }

    // Eliminar usuario (CASCADE elimina todo)
    await prisma.user.delete({
      where: { id: session.user.id },
    })

    // Cerrar sesion
    await signOut({ redirect: false })

    return NextResponse.json({ success: true })
  } catch (error) {
    if (error instanceof z.ZodError) {
      return NextResponse.json({ error: error.errors[0].message }, { status: 400 })
    }
    return NextResponse.json({ error: "Error al eliminar cuenta" }, { status: 500 })
  }
}
```

## Componente: Delete Account Modal

```typescript
// components/settings/delete-account-modal.tsx
"use client"

import { useState } from "react"
import { useRouter } from "next/navigation"
import {
  AlertDialog,
  AlertDialogAction,
  AlertDialogCancel,
  AlertDialogContent,
  AlertDialogDescription,
  AlertDialogFooter,
  AlertDialogHeader,
  AlertDialogTitle,
  AlertDialogTrigger,
} from "@/components/ui/alert-dialog"
import { Button } from "@/components/ui/button"
import { Input } from "@/components/ui/input"
import { Label } from "@/components/ui/label"
import { Trash2, Download } from "lucide-react"

export function DeleteAccountModal() {
  const router = useRouter()
  const [password, setPassword] = useState("")
  const [confirmation, setConfirmation] = useState("")
  const [isLoading, setIsLoading] = useState(false)
  const [error, setError] = useState("")

  async function handleExport() {
    const response = await fetch("/api/user/export")
    const blob = await response.blob()
    const url = URL.createObjectURL(blob)
    const a = document.createElement("a")
    a.href = url
    a.download = `walletwise-export-${new Date().toISOString().split("T")[0]}.json`
    a.click()
    URL.revokeObjectURL(url)
  }

  async function handleDelete() {
    setError("")
    setIsLoading(true)

    try {
      const response = await fetch("/api/user/delete", {
        method: "POST",
        body: JSON.stringify({ password, confirmation }),
      })

      const result = await response.json()

      if (!response.ok) {
        setError(result.error)
        return
      }

      // Redirect a landing
      router.push("/?deleted=true")
    } catch {
      setError("Error al eliminar cuenta")
    } finally {
      setIsLoading(false)
    }
  }

  const isConfirmationValid = confirmation === "ELIMINAR MI CUENTA"

  return (
    <AlertDialog>
      <AlertDialogTrigger asChild>
        <Button variant="destructive">
          <Trash2 className="mr-2 h-4 w-4" />
          Eliminar cuenta
        </Button>
      </AlertDialogTrigger>
      <AlertDialogContent>
        <AlertDialogHeader>
          <AlertDialogTitle>Eliminar cuenta permanentemente</AlertDialogTitle>
          <AlertDialogDescription>
            Esta accion no se puede deshacer. Se eliminaran todos tus datos,
            incluyendo cuentas, transacciones, presupuestos e inventario.
          </AlertDialogDescription>
        </AlertDialogHeader>

        <div className="space-y-4 py-4">
          {/* Export option */}
          <div className="rounded-lg border p-4">
            <p className="text-sm text-muted-foreground mb-2">
              Antes de eliminar, puedes descargar una copia de tus datos:
            </p>
            <Button variant="outline" size="sm" onClick={handleExport}>
              <Download className="mr-2 h-4 w-4" />
              Exportar mis datos
            </Button>
          </div>

          {/* Password */}
          <div className="space-y-2">
            <Label htmlFor="password">Confirma tu password</Label>
            <Input
              id="password"
              type="password"
              value={password}
              onChange={(e) => setPassword(e.target.value)}
              placeholder="Tu password actual"
            />
          </div>

          {/* Confirmation text */}
          <div className="space-y-2">
            <Label htmlFor="confirmation">
              Escribe <span className="font-mono">ELIMINAR MI CUENTA</span> para confirmar
            </Label>
            <Input
              id="confirmation"
              value={confirmation}
              onChange={(e) => setConfirmation(e.target.value)}
              placeholder="ELIMINAR MI CUENTA"
            />
          </div>

          {error && (
            <p className="text-sm text-destructive">{error}</p>
          )}
        </div>

        <AlertDialogFooter>
          <AlertDialogCancel>Cancelar</AlertDialogCancel>
          <AlertDialogAction
            onClick={handleDelete}
            disabled={!isConfirmationValid || !password || isLoading}
            className="bg-destructive text-destructive-foreground hover:bg-destructive/90"
          >
            {isLoading ? "Eliminando..." : "Eliminar cuenta"}
          </AlertDialogAction>
        </AlertDialogFooter>
      </AlertDialogContent>
    </AlertDialog>
  )
}
```
