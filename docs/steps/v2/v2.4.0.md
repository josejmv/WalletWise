# WalletWise v2.4.0 - E2E Encryption

> Encriptacion de extremo a extremo para datos sensibles

**Estado:** Planificado
**Tipo:** MINOR
**Prerequisito:** v2.3.0

---

## Objetivo

Implementar encriptacion de extremo a extremo (E2E) para datos financieros sensibles, de forma que el servidor nunca tenga acceso a los datos en texto plano.

---

## Estrategia de Encriptacion

### Flujo General

```
1. Usuario crea cuenta con password
2. Se deriva una Master Key del password usando PBKDF2
3. Master Key se almacena SOLO en el cliente (memoria/localStorage encriptado)
4. Datos sensibles se encriptan con AES-GCM antes de enviar al servidor
5. El servidor almacena datos encriptados
6. El cliente desencripta al recibir datos
```

### Algoritmos

| Proposito | Algoritmo | Parametros |
|-----------|-----------|------------|
| Key Derivation | PBKDF2 | SHA-256, 100,000 iterations |
| Encryption | AES-GCM | 256 bits |
| IV | Random | 12 bytes por operacion |

## Implementacion

### Key Derivation (PBKDF2)

```typescript
// lib/crypto/key-derivation.ts
export async function deriveKey(
  password: string,
  salt: Uint8Array
): Promise<CryptoKey> {
  const encoder = new TextEncoder()
  const passwordBuffer = encoder.encode(password)

  // Importar password como raw key
  const baseKey = await crypto.subtle.importKey(
    "raw",
    passwordBuffer,
    "PBKDF2",
    false,
    ["deriveBits", "deriveKey"]
  )

  // Derivar key con PBKDF2
  return crypto.subtle.deriveKey(
    {
      name: "PBKDF2",
      salt,
      iterations: 100000,
      hash: "SHA-256",
    },
    baseKey,
    { name: "AES-GCM", length: 256 },
    true, // extractable para backup
    ["encrypt", "decrypt"]
  )
}

export function generateSalt(): Uint8Array {
  return crypto.getRandomValues(new Uint8Array(16))
}
```

### Encryption (AES-GCM)

```typescript
// lib/crypto/encryption.ts
export async function encrypt(
  data: string,
  key: CryptoKey
): Promise<EncryptedField> {
  const encoder = new TextEncoder()
  const iv = crypto.getRandomValues(new Uint8Array(12))

  const ciphertext = await crypto.subtle.encrypt(
    { name: "AES-GCM", iv },
    key,
    encoder.encode(data)
  )

  return {
    ciphertext: bufferToBase64(ciphertext),
    iv: bufferToBase64(iv),
    isEncrypted: true,
  }
}

export async function decrypt(
  encrypted: EncryptedField,
  key: CryptoKey
): Promise<string> {
  const decoder = new TextDecoder()

  const plaintext = await crypto.subtle.decrypt(
    { name: "AES-GCM", iv: base64ToBuffer(encrypted.iv) },
    key,
    base64ToBuffer(encrypted.ciphertext)
  )

  return decoder.decode(plaintext)
}

// Helpers
function bufferToBase64(buffer: ArrayBuffer): string {
  return btoa(String.fromCharCode(...new Uint8Array(buffer)))
}

function base64ToBuffer(base64: string): ArrayBuffer {
  const binary = atob(base64)
  const bytes = new Uint8Array(binary.length)
  for (let i = 0; i < binary.length; i++) {
    bytes[i] = binary.charCodeAt(i)
  }
  return bytes.buffer
}
```

### Key Storage

```typescript
// lib/crypto/key-storage.ts
const KEY_STORAGE_NAME = "walletwise_master_key"

export async function storeMasterKey(key: CryptoKey): Promise<void> {
  // Exportar key
  const exported = await crypto.subtle.exportKey("raw", key)
  const base64 = bufferToBase64(exported)

  // Almacenar en sessionStorage (se pierde al cerrar tab)
  sessionStorage.setItem(KEY_STORAGE_NAME, base64)
}

export async function getMasterKey(): Promise<CryptoKey | null> {
  const base64 = sessionStorage.getItem(KEY_STORAGE_NAME)
  if (!base64) return null

  const buffer = base64ToBuffer(base64)

  return crypto.subtle.importKey(
    "raw",
    buffer,
    { name: "AES-GCM", length: 256 },
    true,
    ["encrypt", "decrypt"]
  )
}

export function clearMasterKey(): void {
  sessionStorage.removeItem(KEY_STORAGE_NAME)
}
```

## Contexto de Encriptacion

```typescript
// contexts/encryption-context.tsx
"use client"

import { createContext, useContext, useState, useEffect, ReactNode } from "react"
import { deriveKey, generateSalt } from "@/lib/crypto/key-derivation"
import { encrypt, decrypt, EncryptedField } from "@/lib/crypto/encryption"
import { storeMasterKey, getMasterKey, clearMasterKey } from "@/lib/crypto/key-storage"

interface EncryptionContextType {
  isUnlocked: boolean
  unlock: (password: string, salt: string) => Promise<boolean>
  lock: () => void
  encryptField: (value: string) => Promise<EncryptedField>
  decryptField: (encrypted: EncryptedField) => Promise<string>
  encryptObject: <T>(obj: T, fields: (keyof T)[]) => Promise<T>
  decryptObject: <T>(obj: T, fields: (keyof T)[]) => Promise<T>
}

const EncryptionContext = createContext<EncryptionContextType | null>(null)

export function EncryptionProvider({ children }: { children: ReactNode }) {
  const [masterKey, setMasterKey] = useState<CryptoKey | null>(null)

  // Intentar recuperar key de sessionStorage al cargar
  useEffect(() => {
    getMasterKey().then(setMasterKey)
  }, [])

  async function unlock(password: string, saltBase64: string): Promise<boolean> {
    try {
      const salt = base64ToBuffer(saltBase64)
      const key = await deriveKey(password, new Uint8Array(salt))
      await storeMasterKey(key)
      setMasterKey(key)
      return true
    } catch {
      return false
    }
  }

  function lock() {
    clearMasterKey()
    setMasterKey(null)
  }

  async function encryptField(value: string): Promise<EncryptedField> {
    if (!masterKey) throw new Error("Encryption locked")
    return encrypt(value, masterKey)
  }

  async function decryptField(encrypted: EncryptedField): Promise<string> {
    if (!masterKey) throw new Error("Encryption locked")
    return decrypt(encrypted, masterKey)
  }

  async function encryptObject<T extends object>(
    obj: T,
    fields: (keyof T)[]
  ): Promise<T> {
    const result = { ...obj }
    for (const field of fields) {
      const value = obj[field]
      if (value !== null && value !== undefined) {
        result[field] = await encryptField(String(value)) as T[keyof T]
      }
    }
    return result
  }

  async function decryptObject<T extends object>(
    obj: T,
    fields: (keyof T)[]
  ): Promise<T> {
    const result = { ...obj }
    for (const field of fields) {
      const value = obj[field] as unknown as EncryptedField
      if (value?.isEncrypted) {
        result[field] = await decryptField(value) as T[keyof T]
      }
    }
    return result
  }

  return (
    <EncryptionContext.Provider
      value={{
        isUnlocked: !!masterKey,
        unlock,
        lock,
        encryptField,
        decryptField,
        encryptObject,
        decryptObject,
      }}
    >
      {children}
    </EncryptionContext.Provider>
  )
}

export function useEncryption() {
  const context = useContext(EncryptionContext)
  if (!context) {
    throw new Error("useEncryption must be used within EncryptionProvider")
  }
  return context
}
```

## Hook de Encriptacion

```typescript
// hooks/use-encryption.ts
import { useEncryption } from "@/contexts/encryption-context"

// Campos encriptados por entidad
const ENCRYPTED_FIELDS = {
  account: ["name", "balance"],
  job: ["name", "description", "payAmount"],
  income: ["description", "amount", "customRate"],
  expense: ["description", "amount", "customRate"],
  transfer: ["description", "amount"],
  budget: ["name", "description", "targetAmount", "currentAmount"],
  inventoryItem: ["name", "notes", "estimatedPrice"],
  category: ["name"],
} as const

type EntityType = keyof typeof ENCRYPTED_FIELDS

export function useEntityEncryption<T extends object>(entityType: EntityType) {
  const { encryptObject, decryptObject } = useEncryption()
  const fields = ENCRYPTED_FIELDS[entityType] as (keyof T)[]

  return {
    encrypt: (obj: T) => encryptObject(obj, fields),
    decrypt: (obj: T) => decryptObject(obj, fields),
    encryptMany: (objs: T[]) => Promise.all(objs.map(obj => encryptObject(obj, fields))),
    decryptMany: (objs: T[]) => Promise.all(objs.map(obj => decryptObject(obj, fields))),
  }
}
```

## Flujo de Registro con Encryption

```
1. Usuario ingresa password
2. Generar salt aleatorio
3. Derivar Master Key con PBKDF2(password, salt)
4. Guardar salt en DB (User.encryptionKeySalt)
5. Almacenar Master Key en sessionStorage
6. Usuario puede crear datos encriptados
```

```typescript
// Registro con encryption
async function handleRegister(email: string, password: string) {
  // Generar salt
  const salt = generateSalt()
  const saltBase64 = bufferToBase64(salt)

  // Registrar usuario
  await fetch("/api/auth/register", {
    method: "POST",
    body: JSON.stringify({
      email,
      password,
      encryptionKeySalt: saltBase64,
    }),
  })

  // Derivar y almacenar key
  const key = await deriveKey(password, salt)
  await storeMasterKey(key)
}
```

## Flujo de Login con Encryption

```
1. Usuario ingresa password
2. Obtener salt del servidor
3. Derivar Master Key con PBKDF2(password, salt)
4. Almacenar Master Key en sessionStorage
5. Usuario puede acceder a datos encriptados
```

```typescript
// Login con encryption
async function handleLogin(email: string, password: string) {
  // Login normal
  const result = await signIn("credentials", { email, password })

  if (result?.ok) {
    // Obtener salt del usuario
    const userRes = await fetch("/api/user/encryption-salt")
    const { salt } = await userRes.json()

    // Derivar y almacenar key
    const key = await deriveKey(password, base64ToBuffer(salt))
    await storeMasterKey(key)
  }
}
```

## Recovery Key

### Generar Recovery Key

```typescript
async function generateRecoveryKey(masterKey: CryptoKey): Promise<string> {
  // Exportar master key
  const exported = await crypto.subtle.exportKey("raw", masterKey)

  // Generar recovery key aleatoria
  const recoveryKey = crypto.getRandomValues(new Uint8Array(32))
  const recoveryKeyBase64 = bufferToBase64(recoveryKey)

  // Encriptar master key con recovery key
  const recoveryEncryptionKey = await crypto.subtle.importKey(
    "raw",
    recoveryKey,
    { name: "AES-GCM", length: 256 },
    false,
    ["encrypt"]
  )

  const iv = crypto.getRandomValues(new Uint8Array(12))
  const encryptedMasterKey = await crypto.subtle.encrypt(
    { name: "AES-GCM", iv },
    recoveryEncryptionKey,
    exported
  )

  // Guardar en servidor
  await fetch("/api/user/recovery-key", {
    method: "POST",
    body: JSON.stringify({
      encryptedMasterKey: bufferToBase64(encryptedMasterKey),
      iv: bufferToBase64(iv),
    }),
  })

  // Retornar recovery key para que el usuario la guarde
  return recoveryKeyBase64
}
```

### Recuperar con Recovery Key

```typescript
async function recoverWithRecoveryKey(recoveryKeyBase64: string): Promise<CryptoKey> {
  // Obtener master key encriptada del servidor
  const res = await fetch("/api/user/recovery-key")
  const { encryptedMasterKey, iv } = await res.json()

  // Derivar key de recovery
  const recoveryKey = base64ToBuffer(recoveryKeyBase64)
  const recoveryEncryptionKey = await crypto.subtle.importKey(
    "raw",
    recoveryKey,
    { name: "AES-GCM", length: 256 },
    false,
    ["decrypt"]
  )

  // Desencriptar master key
  const masterKeyBuffer = await crypto.subtle.decrypt(
    { name: "AES-GCM", iv: base64ToBuffer(iv) },
    recoveryEncryptionKey,
    base64ToBuffer(encryptedMasterKey)
  )

  // Importar master key
  return crypto.subtle.importKey(
    "raw",
    masterKeyBuffer,
    { name: "AES-GCM", length: 256 },
    true,
    ["encrypt", "decrypt"]
  )
}
```

## Consideraciones de Seguridad

### Recomendaciones

- [ ] Master Key nunca se envia al servidor
- [ ] Salt es unico por usuario
- [ ] IV es unico por operacion de encriptacion
- [ ] Recovery Key solo se muestra una vez al usuario
- [ ] sessionStorage se limpia al cerrar tab
- [ ] Logout limpia sessionStorage

### Limitaciones

| Limitacion | Descripcion |
|------------|-------------|
| Busqueda | No se puede buscar en campos encriptados |
| Ordenamiento | No se puede ordenar por campos encriptados |
| Calculos en servidor | Totales/agregados deben hacerse en cliente |
| Olvido de password | Requiere recovery key |
