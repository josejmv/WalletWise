# WalletWise v2.3.0 - Multi-Usuario

> Migracion de datos y aislamiento por usuario

**Estado:** Planificado
**Tipo:** MINOR
**Prerequisito:** v2.2.0

---

## Objetivo

Migrar todas las entidades existentes para soportar multiples usuarios con aislamiento completo de datos.

## Helper de Autenticacion

### lib/api-auth.ts

```typescript
import { auth } from "@/lib/auth"
import { NextResponse } from "next/server"

export async function withAuth<T>(
  handler: (userId: string, request: Request) => Promise<T>
) {
  return async (request: Request) => {
    const session = await auth()

    if (!session?.user?.id) {
      return NextResponse.json(
        { success: false, error: "No autorizado" },
        { status: 401 }
      )
    }

    return handler(session.user.id, request)
  }
}

// Uso
export const GET = withAuth(async (userId, request) => {
  const expenses = await getExpenses(userId)
  return NextResponse.json({ success: true, data: expenses })
})
```

## Script de Migracion

### prisma/migrations/migrate-to-multiuser.ts

```typescript
import { PrismaClient } from "@prisma/client"

const prisma = new PrismaClient()

async function main() {
  // Buscar el usuario principal (el primero creado o uno especifico)
  const mainUser = await prisma.user.findFirst({
    orderBy: { createdAt: "asc" },
  })

  if (!mainUser) {
    console.log("No hay usuarios. Ejecuta el seed primero.")
    return
  }

  console.log(`Migrando datos al usuario: ${mainUser.email}`)

  // Migrar todas las entidades al usuario principal
  const tables = [
    "account",
    "job",
    "income",
    "expense",
    "transfer",
    "budget",
    "budgetContribution",
    "category",
    "inventoryCategory",
    "inventoryItem",
    "inventoryPriceHistory",
  ]

  for (const table of tables) {
    const result = await prisma.$executeRawUnsafe(
      `UPDATE "${table}" SET "userId" = $1 WHERE "userId" IS NULL`,
      mainUser.id
    )
    console.log(`${table}: ${result} registros actualizados`)
  }

  // UserConfig - actualizar o crear
  const existingConfig = await prisma.userConfig.findFirst()
  if (existingConfig) {
    await prisma.userConfig.update({
      where: { id: existingConfig.id },
      data: { userId: mainUser.id },
    })
    console.log("UserConfig actualizado")
  }

  console.log("Migracion completada")
}

main()
  .catch(console.error)
  .finally(() => prisma.$disconnect())
```

) {
  const session = await auth()
