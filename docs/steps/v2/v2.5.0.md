### Spotlight

```typescript
// components/onboarding/spotlight.tsx
"use client"

import { useEffect, useState, ReactNode } from "react"
import { createPortal } from "react-dom"

interface SpotlightProps {
  targetSelector: string
  isActive: boolean
  children: ReactNode
  position?: "top" | "bottom" | "left" | "right"
}

export function Spotlight({
  targetSelector,
  isActive,
  children,
  position = "bottom",
}: SpotlightProps) {
  const [targetRect, setTargetRect] = useState<DOMRect | null>(null)

  useEffect(() => {
    if (!isActive) return

    const target = document.querySelector(targetSelector)
    if (target) {
      setTargetRect(target.getBoundingClientRect())
    }
  }, [targetSelector, isActive])

  if (!isActive || !targetRect) return null

  const overlayStyle = {
    position: "fixed" as const,
    top: 0,
    left: 0,
    right: 0,
    bottom: 0,
    zIndex: 50,
  }

  const spotlightStyle = {
    position: "absolute" as const,
    top: targetRect.top - 8,
    left: targetRect.left - 8,
    width: targetRect.width + 16,
    height: targetRect.height + 16,
    borderRadius: 8,
    boxShadow: "0 0 0 9999px rgba(0, 0, 0, 0.7)",
  }

  const tooltipStyle = {
    position: "absolute" as const,
    ...getTooltipPosition(targetRect, position),
  }

  return createPortal(
    <div style={overlayStyle}>
      <div style={spotlightStyle} />
      <div style={tooltipStyle}>{children}</div>
    </div>,
    document.body
  )
}

function getTooltipPosition(rect: DOMRect, position: string) {
  switch (position) {
    case "bottom":
      return { top: rect.bottom + 16, left: rect.left }
    case "top":
      return { bottom: window.innerHeight - rect.top + 16, left: rect.left }
    case "right":
      return { top: rect.top, left: rect.right + 16 }
    case "left":
      return { top: rect.top, right: window.innerWidth - rect.left + 16 }
    default:
      return { top: rect.bottom + 16, left: rect.left }
  }
}
```

```typescript
import { Card, CardContent, CardHeader, CardTitle } from "@/components/ui/card"
import { Button } from "@/components/ui/button"
import { X } from "lucide-react"

interface TooltipCardProps {
  title: string
  description: string
  step?: { current: number; total: number }
  onNext?: () => void
  onPrev?: () => void
  onClose?: () => void
  nextLabel?: string
  prevLabel?: string
}

export function TooltipCard({
  title,
  description,
  step,
  onNext,
  onPrev,
  onClose,
  nextLabel = "Siguiente",
  prevLabel = "Anterior",
}: TooltipCardProps) {
  return (
    <Card className="w-80 shadow-lg">
      <CardHeader className="pb-2">
        <div className="flex items-start justify-between">
          <CardTitle className="text-lg">{title}</CardTitle>
          {onClose && (
            <Button variant="ghost" size="icon" onClick={onClose}>
              <X className="h-4 w-4" />
            </Button>
          )}
        </div>
        {step && (
          <p className="text-xs text-muted-foreground">
            Paso {step.current} de {step.total}
          </p>
        )}
      </CardHeader>
      <CardContent className="space-y-4">
        <p className="text-sm text-muted-foreground">{description}</p>
        <div className="flex justify-between">
          {onPrev ? (
            <Button variant="outline" size="sm" onClick={onPrev}>
              {prevLabel}
            </Button>
          ) : (
            <div />
          )}
          {onNext && (
            <Button size="sm" onClick={onNext}>
              {nextLabel}
            </Button>
          )}
        </div>
      </CardContent>
    </Card>
  )
}
```

### Progress Checklist

```typescript
// components/onboarding/progress-checklist.tsx
"use client"

import { useOnboarding } from "@/hooks/use-onboarding"
import { Card, CardContent, CardHeader, CardTitle } from "@/components/ui/card"
import { Progress } from "@/components/ui/progress"
import { Check, Circle } from "lucide-react"
import Link from "next/link"

const STEPS = [
  { key: "accountsCompleted", label: "Crear cuenta", href: "/dashboard/accounts" },
  { key: "categoriesCompleted", label: "Crear categoria", href: "/dashboard/categories" },
  { key: "firstIncomeCompleted", label: "Registrar ingreso", href: "/dashboard/incomes" },
  { key: "firstExpenseCompleted", label: "Registrar gasto", href: "/dashboard/expenses" },
  { key: "budgetCompleted", label: "Crear presupuesto", href: "/dashboard/budgets" },
  { key: "dashboardTourCompleted", label: "Explorar dashboard", href: "/dashboard" },
]

export function ProgressChecklist() {
  const { progress, isComplete } = useOnboarding()

  if (!progress || isComplete) return null

  const completedCount = STEPS.filter(
    step => progress[step.key as keyof typeof progress]
  ).length
  const progressPercent = (completedCount / STEPS.length) * 100

  return (
    <Card>
      <CardHeader>
        <CardTitle className="text-lg">Primeros Pasos</CardTitle>
        <Progress value={progressPercent} className="h-2" />
        <p className="text-xs text-muted-foreground">
          {completedCount} de {STEPS.length} completados
        </p>
      </CardHeader>
      <CardContent>
        <ul className="space-y-2">
          {STEPS.map(step => {
            const isCompleted = progress[step.key as keyof typeof progress]
            return (
              <li key={step.key}>
                <Link
                  href={step.href}
                  className={`flex items-center gap-2 text-sm ${
                    isCompleted ? "text-muted-foreground line-through" : ""
                  }`}
                >
                  {isCompleted ? (
                    <Check className="h-4 w-4 text-green-500" />
                  ) : (
                    <Circle className="h-4 w-4" />
                  )}
                  {step.label}
                </Link>
              </li>
            )
          })}
        </ul>
      </CardContent>
    </Card>
  )
}
```

### Mini Tour

```typescript
// components/onboarding/mini-tour.tsx
"use client"

import { useState } from "react"
import { Spotlight } from "./spotlight"
import { TooltipCard } from "./tooltip-card"
import { useOnboarding } from "@/hooks/use-onboarding"

interface TourStep {
  targetSelector: string
  title: string
  description: string
  position?: "top" | "bottom" | "left" | "right"
}

interface MiniTourProps {
  tourKey: string
  steps: TourStep[]
}

export function MiniTour({ tourKey, steps }: MiniTourProps) {
  const { progress, completeStep } = useOnboarding()
  const [currentStep, setCurrentStep] = useState(0)

  // Ya completo este tour
  const tourKeyComplete = `${tourKey}TourCompleted` as keyof typeof progress
  if (progress?.[tourKeyComplete]) return null

  const step = steps[currentStep]
  const isLast = currentStep === steps.length - 1

  async function handleNext() {
    if (isLast) {
      await completeStep(tourKeyComplete)
    } else {
      setCurrentStep(prev => prev + 1)
    }
  }

  function handlePrev() {
    setCurrentStep(prev => Math.max(0, prev - 1))
  }

  async function handleClose() {
    await completeStep(tourKeyComplete)
  }

  return (
    <Spotlight
      targetSelector={step.targetSelector}
      isActive={true}
      position={step.position}
    >
      <TooltipCard
        title={step.title}
        description={step.description}
        step={{ current: currentStep + 1, total: steps.length }}
        onNext={handleNext}
        onPrev={currentStep > 0 ? handlePrev : undefined}
        onClose={handleClose}
        nextLabel={isLast ? "Finalizar" : "Siguiente"}
      />
    </Spotlight>
  )
}
```

## Contexto de Onboarding

```typescript
// contexts/onboarding-context.tsx
"use client"

import { createContext, useContext, useState, useEffect, ReactNode } from "react"
import { useSession } from "next-auth/react"

interface OnboardingProgress {
  welcomeCompleted: boolean
  accountsCompleted: boolean
  categoriesCompleted: boolean
  firstIncomeCompleted: boolean
  firstExpenseCompleted: boolean
  budgetCompleted: boolean
  dashboardTourCompleted: boolean
  accountsTourCompleted: boolean
  expensesTourCompleted: boolean
  incomesTourCompleted: boolean
  budgetsTourCompleted: boolean
  transfersTourCompleted: boolean
  inventoryTourCompleted: boolean
  reportsTourCompleted: boolean
  completedAt: string | null
  skippedAt: string | null
}

interface OnboardingContextType {
  progress: OnboardingProgress | null
  isLoading: boolean
  isComplete: boolean
  completeStep: (step: keyof OnboardingProgress, value?: unknown) => Promise<void>
  resetOnboarding: () => Promise<void>
}

const OnboardingContext = createContext<OnboardingContextType | null>(null)

export function OnboardingProvider({ children }: { children: ReactNode }) {
  const { data: session } = useSession()
  const [progress, setProgress] = useState<OnboardingProgress | null>(null)
  const [isLoading, setIsLoading] = useState(true)

  useEffect(() => {
    if (session?.user?.id) {
      fetchProgress()
    }
  }, [session])

  async function fetchProgress() {
    try {
      const res = await fetch("/api/onboarding")
      const data = await res.json()
      setProgress(data)
    } catch (error) {
      console.error("Error fetching onboarding:", error)
    } finally {
      setIsLoading(false)
    }
  }

  async function completeStep(step: keyof OnboardingProgress, value: unknown = true) {
    const res = await fetch("/api/onboarding", {
      method: "POST",
      body: JSON.stringify({ [step]: value }),
    })
    const updated = await res.json()
    setProgress(updated)
  }

  async function resetOnboarding() {
    await fetch("/api/onboarding/reset", { method: "POST" })
    await fetchProgress()
  }

  const isComplete = !!(progress?.completedAt || progress?.skippedAt)

  return (
    <OnboardingContext.Provider
      value={{ progress, isLoading, isComplete, completeStep, resetOnboarding }}
    >
      {children}
    </OnboardingContext.Provider>
  )
}

export function useOnboarding() {
  const context = useContext(OnboardingContext)
  if (!context) {
    throw new Error("useOnboarding must be used within OnboardingProvider")
  }
  return context
}
```

## API Routes

### GET/POST Progress

```typescript
// app/api/onboarding/route.ts
import { auth } from "@/lib/auth"
import { prisma } from "@/lib/prisma"
import { NextResponse } from "next/server"

export async function GET() {
  const session = await auth()
  if (!session?.user?.id) {
    return NextResponse.json({ error: "No autorizado" }, { status: 401 })
  }

  let progress = await prisma.onboardingProgress.findUnique({
    where: { userId: session.user.id },
  })

  // Crear si no existe
  if (!progress) {
    progress = await prisma.onboardingProgress.create({
      data: { userId: session.user.id },
    })
  }

  return NextResponse.json(progress)
}

export async function POST(request: Request) {
  const session = await auth()
  if (!session?.user?.id) {
    return NextResponse.json({ error: "No autorizado" }, { status: 401 })
  }

  const updates = await request.json()

  const progress = await prisma.onboardingProgress.upsert({
    where: { userId: session.user.id },
    create: { userId: session.user.id, ...updates },
    update: updates,
  })

  // Verificar si se completo todo
  const allDone =
    progress.accountsCompleted &&
    progress.categoriesCompleted &&
    progress.firstIncomeCompleted &&
    progress.firstExpenseCompleted &&
    progress.budgetCompleted &&
    progress.dashboardTourCompleted

  if (allDone && !progress.completedAt) {
    await prisma.onboardingProgress.update({
      where: { userId: session.user.id },
      data: { completedAt: new Date() },
    })
  }

  return NextResponse.json(progress)
}
```

### Reset Onboarding

```typescript
// app/api/onboarding/reset/route.ts
import { auth } from "@/lib/auth"
import { prisma } from "@/lib/prisma"
import { NextResponse } from "next/server"

export async function POST() {
  const session = await auth()
  if (!session?.user?.id) {
    return NextResponse.json({ error: "No autorizado" }, { status: 401 })
  }

  await prisma.onboardingProgress.update({
    where: { userId: session.user.id },
    data: {
      welcomeCompleted: false,
      accountsCompleted: false,
      categoriesCompleted: false,
      firstIncomeCompleted: false,
      firstExpenseCompleted: false,
      budgetCompleted: false,
      dashboardTourCompleted: false,
      accountsTourCompleted: false,
      expensesTourCompleted: false,
      incomesTourCompleted: false,
      budgetsTourCompleted: false,
      transfersTourCompleted: false,
      inventoryTourCompleted: false,
      reportsTourCompleted: false,
      completedAt: null,
      skippedAt: null,
    },
  })

  return NextResponse.json({ success: true })
}
```
